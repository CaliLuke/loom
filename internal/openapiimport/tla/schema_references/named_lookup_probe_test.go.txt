package openapiimport

import (
	"strings"
	"testing"

	"github.com/stretchr/testify/require"
)

func TestRecursiveNamedLookupReadinessControl(t *testing.T) {
	cases := []struct {
		name    string
		schemas string
	}{
		{"acyclic", `Node: {type: object, properties: {name: {type: string}}}`},
		{"self_array", `Node:
      type: object
      properties:
        name: {type: string}
        children: {type: array, items: {$ref: '#/components/schemas/Node'}}`},
		{"mutual_map_array", `Node:
      type: object
      properties:
        branches: {type: object, additionalProperties: {$ref: '#/components/schemas/Branch'}}
    Branch:
      type: object
      properties:
        children: {type: array, items: {$ref: '#/components/schemas/Node'}}`},
		{"mixed_inheritance", `Node:
      allOf:
        - {$ref: '#/components/schemas/Base'}
        - type: object
          properties: {name: {type: string}}
    Base:
      type: object
      properties:
        child: {$ref: '#/components/schemas/Node'}`},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			source := []byte(`openapi: 3.1.1
info: {title: Recursive, version: "1"}
paths:
  /nodes:
    get:
      operationId: getNodes
      responses:
        "200":
          description: Node
          content:
            application/json:
              schema: {$ref: '#/components/schemas/Node'}
components:
  schemas:
    ` + tc.schemas + "\n")
			doc, diagnostics, err := Analyze(source)
			require.NoError(t, err)
			if tc.name == "mixed_inheritance" {
				require.Len(t, diagnostics, 1)
				requireDiagnosticCode(t, diagnostics, "schema-allof-flattened")
				fatal, _ := diagnostics.Classify(true)
				require.Empty(t, fatal)
			} else {
				require.Empty(t, diagnostics)
			}
			design, err := Render(doc, Options{PackageName: "design"})
			require.NoError(t, err)
			candidate := string(design)
			candidate = strings.ReplaceAll(candidate, "ArrayOf(ImportedNode)", `ArrayOf("Node")`)
			candidate = strings.ReplaceAll(candidate, "MapOf(String, ImportedBranch)", `MapOf(String, "Branch")`)
			candidate = strings.ReplaceAll(candidate, `Attribute("child", ImportedNode)`, `Attribute("child", "Node")`)
			requireRenderedDesignEvaluates(t, []byte(candidate), 1)
		})
	}
}
