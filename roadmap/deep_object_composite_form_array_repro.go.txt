package expr_test

import (
	"testing"

	"github.com/stretchr/testify/require"

	. "github.com/CaliLuke/loom/dsl"
	"github.com/CaliLuke/loom/expr"
)

func TestDeepObjectReadinessUnsupportedFormArray(t *testing.T) {
	for _, kind := range []string{"object", "map"} {
		t.Run(kind, func(t *testing.T) {
			err := expr.RunInvalidDSL(t, func() {
				var element expr.DataType = MapOf(String, String)
				if kind == "object" {
					element = Type("Address", func() {
						Attribute("street", String)
					})
				}
				Service("form", func() {
					Method("submit", func() {
						Payload(func() {
							Attribute("items", ArrayOf(element))
						})
						HTTP(func() {
							POST("/")
							FormRequest()
						})
					})
				})
			})
			require.NotNil(t, err, "the form runtime cannot encode composite array elements")
		})
	}
}
