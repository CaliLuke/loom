			Method("selected", func() {
				Error("oops")
				HTTP(func() {
					GET("/selected")
					Response("oops", StatusBadRequest, func() {

	runGoCommand(t, dir, "mod", "tidy")
	runGoCommand(t, dir, "build", "./...")
}
